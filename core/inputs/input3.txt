#include <stdio.h>

int main() {
    int i = 5;
    float f = 2.5;
    float r1 = i + f;  
    float r2 = i + 1.2
    float t = (i > 3) ? i : f;

    printf("=== Promociones en printf ===\n");
    printf("float %f\n", f);        
    
    printf("\n=== Usual arithmetic conversions ===\n");
    printf("int + float = %f\n", r1);          
    printf("int + float = %f\n", r2);  


    printf("t = %f\n", t);      

    printf("%f\n", (i > 3) ? i : f);
    printf("%f\n", (i > 3) ? f : i);
    return 0;
}
