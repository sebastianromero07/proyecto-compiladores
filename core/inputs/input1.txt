#include <stdio.h>

int main() {
    int a = 10;
    unsigned int b = 20;
    float c = 3.5;
    float d = 2.0;

    printf("int + int = %d\n", a + a);
    printf("int - int = %d\n", a - a);
    printf("int * int = %d\n", a * a);
    printf("int / int = %d\n", a / a);

    printf("unsigned + unsigned = %u\n", b + b);
    printf("unsigned - unsigned = %u\n", b - b);
    printf("unsigned * unsigned = %u\n", b * b);
    printf("unsigned / unsigned = %u\n", b / b);

    printf("float + float = %f\n", c + d);
    printf("float - float = %f\n", c - d);
    printf("float * float = %f\n", c * d);
    printf("float / float = %f\n", c / d);

    printf("int + unsigned = %u\n", a + b);
    printf("int - unsigned = %u\n", a - b);
    printf("int * unsigned = %u\n", a * b);
    printf("int / unsigned = %u\n", a / b);

    printf("int + float = %f\n", a + c);
    printf("int - float = %f\n", a - c);
    printf("int * float = %f\n", a * c);
    printf("int / float = %f\n", a / c);

    printf("unsigned + float = %f\n", b + c);
    printf("unsigned - float = %f\n", b - c);
    printf("unsigned * float = %f\n", b * c);
    printf("unsigned / float = %f\n", b / c);

    return 0;
}
